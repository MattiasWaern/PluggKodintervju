8. Felsök en React-komponent KLAR

Någon har skrivit följande komponent:
import { useState } from "react";

export function Counter() {
  const [count, setCount] = useState(0);

  function increment() {
    count = count + 1;
    setCount(count);
  }

  return (
    <div>
      <p>{count}</p>
      <button onClick={increment}>
        Öka
      </button>
    </div>
  );
}
